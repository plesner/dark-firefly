cmake_minimum_required(VERSION 3.8)
set(CMAKE_VERBOSE_MAKEFILE ON)
project(dafi)
add_definitions(-std=c++17)
set (CMAKE_CXX_STANDARD 17)

if (DEFINED EMSCRIPTEN)
  project(emscripten_project)
  set(CMAKE_EXECUTABLE_SUFFIX ".mjs")
  add_executable(engine src/wasm/bindings.cc)
  set_target_properties(engine PROPERTIES LINK_FLAGS "-lembind --emit-tsd engine.d.ts --no-entry")
else()
  add_library(engine SHARED src/wasm/bindings.cc)
endif()

target_include_directories(engine PUBLIC "$ENV{FLATBUFS_ROOT}/include/")
